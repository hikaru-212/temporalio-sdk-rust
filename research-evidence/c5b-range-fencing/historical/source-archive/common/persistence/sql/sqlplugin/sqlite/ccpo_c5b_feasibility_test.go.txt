package sqlite

import (
	"context"
	"database/sql"
	"fmt"
	"os"
	"path/filepath"
	"testing"
	"time"

	"github.com/stretchr/testify/require"
	"go.temporal.io/server/common/config"
	"go.temporal.io/server/common/log"
	"go.temporal.io/server/common/metrics"
	"go.temporal.io/server/common/persistence/sql/sqlplugin"
	"go.temporal.io/server/common/resolver"
	modernc "modernc.org/sqlite"
)

func TestCCPOC5BPhysicalFeasibility(t *testing.T) {
	ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
	defer cancel()
	dir := os.Getenv("CCPO_EVIDENCE_OUTPUT")
	if dir == "" {
		dir = t.TempDir()
	}
	cfg := &config.SQL{DatabaseName: filepath.Join(dir, "feasibility.sqlite"), MaxConns: 2, MaxIdleConns: 2, ConnectAttributes: map[string]string{"cache": "private", "journal_mode": "wal", "read_uncommitted": "0", "busy_timeout": "1000", "synchronous": "normal"}}
	dsn, err := buildDSN(cfg)
	require.NoError(t, err)
	p := &plugin{connPool: newConnPool()}
	generic, err := p.CreateDB(sqlplugin.DbKindMain, cfg, resolver.NewNoopResolver(), log.NewNoopLogger(), metrics.NoopMetricsHandler)
	require.NoError(t, err)
	db := generic.(*db).db
	defer db.Close()
	_, err = db.ExecContext(ctx, "CREATE TABLE ccpo_probe (id INTEGER PRIMARY KEY, value INTEGER)")
	require.NoError(t, err)
	_, err = db.ExecContext(ctx, "INSERT INTO ccpo_probe VALUES (1, 0)")
	require.NoError(t, err)
	a, err := db.Conn(ctx)
	require.NoError(t, err)
	defer a.Close()
	b, err := db.Conn(ctx)
	require.NoError(t, err)
	defer b.Close()
	identities := make([]string, 0, 2)
	for i, conn := range []*sql.Conn{a, b} {
		err = conn.Raw(func(raw any) error {
			identities = append(identities, fmt.Sprintf("%T:%p", raw, raw))
			return nil
		})
		require.NoError(t, err)
		for _, pragma := range []string{"journal_mode", "read_uncommitted", "synchronous", "busy_timeout", "database_list"} {
			rows, queryErr := conn.QueryContext(ctx, "PRAGMA "+pragma)
			require.NoError(t, queryErr)
			columns, colErr := rows.Columns()
			require.NoError(t, colErr)
			for rows.Next() {
				values := make([]any, len(columns))
				ptrs := make([]any, len(columns))
				for j := range values {
					ptrs[j] = &values[j]
				}
				require.NoError(t, rows.Scan(ptrs...))
				t.Logf("PHYSICAL conn=%d pragma=%s columns=%v values=%v", i, pragma, columns, values)
				if pragma == "journal_mode" {
					require.Equal(t, "wal", values[0])
				}
				if pragma == "read_uncommitted" {
					require.EqualValues(t, 0, values[0])
				}
			}
			require.NoError(t, rows.Err())
			require.NoError(t, rows.Close())
		}
	}
	require.NotEqual(t, identities[0], identities[1])
	t.Logf("PHYSICAL dsn=%s identities=%v pool=%+v", dsn, identities, db.Stats())
	var version string
	require.NoError(t, a.QueryRowContext(ctx, "SELECT sqlite_version()").Scan(&version))
	t.Logf("PHYSICAL sqlite_version=%s driver=modernc.org/sqlite@v1.51.0", version)
	tx, err := a.BeginTx(ctx, nil)
	require.NoError(t, err)
	t.Log("PHYSICAL A BEGIN returned; no read yet")
	_, err = b.ExecContext(ctx, "UPDATE ccpo_probe SET value=1 WHERE id=1")
	require.NoError(t, err)
	var value int
	require.NoError(t, tx.QueryRowContext(ctx, "SELECT value FROM ccpo_probe WHERE id=1").Scan(&value))
	require.Equal(t, 1, value)
	t.Log("PHYSICAL first A SELECT establishes snapshot value=1 after B commit1")
	_, err = b.ExecContext(ctx, "UPDATE ccpo_probe SET value=2 WHERE id=1")
	require.NoError(t, err)
	require.NoError(t, tx.QueryRowContext(ctx, "SELECT value FROM ccpo_probe WHERE id=1").Scan(&value))
	require.Equal(t, 1, value)
	t.Log("PHYSICAL B commit2 succeeded; open A snapshot still value=1")
	_, err = tx.ExecContext(ctx, "UPDATE ccpo_probe SET value=3 WHERE id=1")
	var sqliteErr *modernc.Error
	require.ErrorAs(t, err, &sqliteErr)
	t.Logf("PHYSICAL A mutation error=%T %v code=%d", err, err, sqliteErr.Code())
	require.Equal(t, 517, sqliteErr.Code())
	require.NoError(t, tx.Rollback())
	require.NoError(t, b.QueryRowContext(ctx, "SELECT value FROM ccpo_probe WHERE id=1").Scan(&value))
	require.Equal(t, 2, value)
	t.Log("PHYSICAL final value=2; Temporal correctness not tested")
}
